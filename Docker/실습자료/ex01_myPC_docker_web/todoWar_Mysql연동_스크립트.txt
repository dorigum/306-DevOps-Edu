# 컨테이너간의 통신을 위한 네트워크 생성
docker network create web_net

# mysql의 볼륨마운트 생성
docker volume create mysql_vol

# tomcat의 볼륨마운트 생성
docker volume create mytomcat_vol

tomcat 서버 기본 경로  = /usr/local/tomcat/webapps


#host pc의 mysql 서비스를 중지시킨다.

# mysql pull + create + start
#-e TZ=Asia/Seoul: MySQL 서버의 시간대를 한국 시간, .-v /etc/localtime:/etc/localtime:ro: 호스트의 시간 정보를 컨테이너와 공유하여 시스템 시간이 일치하도록 한다.
docker run --name my_sql -e MYSQL_ROOT_PASSWORD=1234 -e MYSQL_DATABASE=web_basic -e TZ=Asia/Seoul -v /etc/localtime:/etc/localtime:ro -d -p 3306:3306 -v mysql_vol:/var/lib/mysql --net web_net mysql:8.4



# tomcat pull + create + start
docker run --name my_tomcat -p 80:8080 -d   -v mytomcat_vol:/usr/local/tomcat/webapps --net web_net tomcat:10

# 프로젝트 ROOT.war 준비하기  - context.xml 설정변경(url, password확인)
/*
 <Resource name="jdbc/mySql"
          auth="Container"
          type="javax.sql.DataSource" 
          driverClassName="com.mysql.cj.jdbc.Driver"
          url="jdbc:mysql://my_sql:3306/web_basic"
          username="root" 
          password="1234" 
          maxTotal="20" 
          maxIdle="10"
          maxWaitMillis="-1"/>
*/

# tomcat기본폴더에 ROOT.war 복사
docker cp ./ROOT.war  my_tomcat:/usr/local/tomcat/webapps

# mysql bash쉘 접속
docker exec –it my_sql /bin/bash 

# mysql로그인 
mysql -uroot -p1234

# 데이터베이스 생성
create database web_basic;

# 데이터베이스선택
use web_basic

# 테이블생성 + 레코드 추가
create table todo_list(
   id int primary key auto_increment,
   done char(1) default 0,
   content varchar(100),
   reg_date datetime default now()
);

insert into todo_list(content) value('Studying');
insert into todo_list(content) value('meeting friends');
insert into todo_list(content) value('Practicing dockers');

select id,done,content, reg_date from todo_list;

commit;

#bash(쉘) 빠져나오기
exit
exit


# tomcat bash쉘 접속
docker exec –it my_tomcat /bin/bash 

# tomcat log정보 확인하기
docker logs -f my_tomcat

# mysql log정보확인하기
docker logs my_sql
